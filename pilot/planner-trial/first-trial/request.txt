Include the required build, unit-test, security-scan, code-review and deployment actions. Deployment must use the same planned artifact as testing and scanning. Satisfy every supplied requirement. Return only JSON conforming to planlens-agent/plan/v1.

===== BEGIN action catalog (catalog.toml) =====
# Action catalog for the release-basic example.
#
# Owned by the application, not by the agent. It lists the actions a plan may
# use, and the named artifact inputs and outputs each one declares.
#   outputs: what the step produces, as  name = "artifact type"
#   inputs:  what the step consumes,  as  name = "accepted artifact type"
# Every declared input must be bound in a plan to an earlier step's output.
#
# This describes actions. It does not show what the code behind them does.

schema = "planlens-agent/catalog/v1"
id = "release-basic"
version = "1.0.0"
description = "Actions for a minimal build, verify, review and deploy release."

[actions.build]
description = "Build a release artifact from the reviewed source."
outputs = { artifact = "release-artifact" }

[actions.unit_tests]
description = "Run unit tests against a built artifact."
inputs = { artifact = "release-artifact" }

[actions.security_scan]
description = "Scan a built artifact for known vulnerabilities."
inputs = { artifact = "release-artifact" }

[actions.code_review]
description = "Human review of the change being released."

[actions.deploy]
description = "Deploy an artifact to production."
inputs = { artifact = "release-artifact" }

[actions.notify]
description = "Announce the release. Optional; the contract does not require it."
===== END action catalog (catalog.toml) =====

===== BEGIN owner contract (contract.toml) =====
# Release contract for the release-basic example.
#
# Owned by the application, not by the agent. A candidate plan cannot remove,
# waive or replace anything here. Every rule is required.

schema = "planlens-agent/contract/v1"
id = "release-basic"
version = "1.0.0"
description = "A release must be built, tested, scanned and reviewed before it is deployed."

# The catalog this contract was written for. A different catalog is refused.
catalog = { id = "release-basic", version = "1.0.0" }

# Every plan must contain at least one step of each of these actions.
required_actions = ["build", "unit_tests", "security_scan", "code_review", "deploy"]

# kind = "prerequisite": every step of `action` needs an earlier step of
# `requires_prior`. Actions with no rule between them may appear in either order.
[[rules]]
id = "review-before-deploy"
kind = "prerequisite"
action = "deploy"
requires_prior = "code_review"
because = "A release may not be deployed before its code review."

[[rules]]
id = "tests-before-deploy"
kind = "prerequisite"
action = "deploy"
requires_prior = "unit_tests"
because = "A release may not be deployed before its unit tests run."

[[rules]]
id = "scan-before-deploy"
kind = "prerequisite"
action = "deploy"
requires_prior = "security_scan"
because = "A release may not be deployed before its security scan runs."

# kind = "same_artifact": these inputs must all refer to one planned artifact,
# meaning the same producer step and output.
[[rules]]
id = "deploy-what-was-verified"
kind = "same_artifact"
inputs = ["unit_tests.artifact", "security_scan.artifact", "deploy.artifact"]
because = "The artifact that is deployed must be the one that was tested and scanned."
===== END owner contract (contract.toml) =====
